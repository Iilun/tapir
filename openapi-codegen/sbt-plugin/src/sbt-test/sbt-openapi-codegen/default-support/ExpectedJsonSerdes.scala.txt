package sttp.tapir.generated

object TapirGeneratedEndpointsJsonSerdes {
  import sttp.tapir.generated.TapirGeneratedEndpoints._
  import sttp.tapir.generic.auto._
  implicit val byteStringJsonDecoder: io.circe.Decoder[ByteString] =
    io.circe.Decoder.decodeString
      .map(java.util.Base64.getDecoder.decode)
      .map(toByteString)
  implicit val byteStringJsonEncoder: io.circe.Encoder[ByteString] =
    io.circe.Encoder.encodeString
      .contramap(java.util.Base64.getEncoder.encodeToString)
  implicit val defaultsConfig: io.circe.generic.extras.Configuration = io.circe.generic.extras.Configuration.default.withDefaults
  implicit lazy val bookJsonDecoder: io.circe.Decoder[Book] = io.circe.generic.extras.semiauto.deriveConfiguredDecoder[Book]
  implicit lazy val bookJsonEncoder: io.circe.Encoder[Book] = io.circe.generic.semiauto.deriveEncoder[Book]
}
